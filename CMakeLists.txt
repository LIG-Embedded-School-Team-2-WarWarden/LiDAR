cmake_minimum_required(VERSION 3.18)
project(warwarden_lidar_capture LANGUAGES CXX)
include(CTest)
option(LIDAR_WITH_XTSDK "Build the real XT SDK capture backend" OFF)
add_library(lidar_capture STATIC capture/simulated.cpp)
target_compile_features(lidar_capture PUBLIC cxx_std_17)
target_include_directories(lidar_capture PUBLIC capture)
if(LIDAR_WITH_XTSDK)
    find_path(XTSDK_INCLUDE_DIR xtsdk.h REQUIRED)
    find_library(XTSDK_LIBRARY NAMES xtsdk REQUIRED)
    target_sources(lidar_capture PRIVATE capture/xt_capture.cpp)
    target_include_directories(lidar_capture PRIVATE "${XTSDK_INCLUDE_DIR}")
    set(XTSDK_EXTRA_LIBRARIES "" CACHE STRING "Libraries required by the supplied XtSdk wrapper (OpenCV, Boost, dl, etc.)")
    target_link_libraries(lidar_capture PUBLIC "${XTSDK_LIBRARY}" ${XTSDK_EXTRA_LIBRARIES})
    target_compile_definitions(lidar_capture PUBLIC LIDAR_WITH_XTSDK)
endif()
if(BUILD_TESTING)
    add_executable(xt_capture_contract tests/capture_contract.cpp capture/xt_capture.cpp)
    target_compile_features(xt_capture_contract PRIVATE cxx_std_17)
    target_compile_definitions(xt_capture_contract PRIVATE LIDAR_WITH_XTSDK)
    target_include_directories(xt_capture_contract PRIVATE capture tests/sdk_shim)
    add_test(NAME xt_capture_contract COMMAND xt_capture_contract)
endif()
